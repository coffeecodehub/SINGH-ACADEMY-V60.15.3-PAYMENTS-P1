/** REAL HTTP + MongoDB replica-set tests. Never substitutes stubs and never silently skips. */
import test from 'node:test';import assert from 'node:assert/strict';import crypto from 'node:crypto';
import {isolatedDatabase,testEnvironment} from './isolation.js';
test('V46 real HTTP authentication and manual billing lifecycle',{timeout:180000},async t=>{
 const isolated=isolatedDatabase();testEnvironment();
 const mongoose=(await import('mongoose')).default,bcrypt=(await import('bcryptjs')).default;
 const {createApp}=await import('../src/app.js');
 const User=(await import('../src/models/User.js')).default,Course=(await import('../src/models/Course.js')).default;
 const Invoice=(await import('../src/models/Invoice.js')).default,Payment=(await import('../src/models/Payment.js')).default,Enrollment=(await import('../src/models/Enrollment.js')).default;
 const PurchaseRequest=(await import('../src/models/PurchaseRequest.js')).default,Refund=(await import('../src/models/Refund.js')).default;
 const {hotp}=await import('../src/utils/totp.js'),{transaction}=await import('../src/services/businessWrite.js');
 await mongoose.connect(isolated.uri,{dbName:isolated.dbName,autoIndex:false,serverSelectionTimeoutMS:12000});
 let server;
 t.after(async()=>{try{if(server){server.closeAllConnections();await new Promise(resolve=>server.close(resolve));}await isolated.cleanup(mongoose);}finally{await mongoose.disconnect();}});
 const hello=await mongoose.connection.db.admin().command({hello:1});assert.ok(hello.setName||hello.msg==='isdbgrid','Replica set/sharded MongoDB required; do not use standalone');
 for(const M of Object.values(mongoose.models))await M.createIndexes();
 const password='Test-'+crypto.randomBytes(20).toString('hex'),passwordHash=await bcrypt.hash(password,12);
 const fixtures={};for(const [key,role]of [['client','client_admin'],['builder','super_admin'],['student','student'],['other','student']])fixtures[key]=await User.create({name:key+' Test Fixture',email:key+'@integration.invalid',role,status:'active',emailVerified:true,passwordHash});
 await Course.create({title:'Integration fixture course',slug:'integration-fixture',published:true,price:100,currency:'USD',accessType:'one_time'});
 server=await new Promise(resolve=>{const s=createApp().listen(0,'127.0.0.1',()=>resolve(s));});const url='http://127.0.0.1:'+server.address().port;
 async function request(route,{method='GET',body,cookie,portal='student',key,headers={}}={}){
  const response=await fetch(url+'/api'+route,{method,headers:{Origin:'http://localhost:3000','X-SA-CSRF':'1','X-SA-Portal':portal,...(body?{'Content-Type':'application/json'}:{}),...(cookie?{Cookie:cookie}:{}),...(key?{'Idempotency-Key':key}:{}),...headers},...(body?{body:JSON.stringify(body)}:{})});
  const raw=await response.text();let data;try{data=JSON.parse(raw);}catch{data={raw};}
  return {status:response.status,data,headers:response.headers,cookie:response.headers.getSetCookie().map(x=>x.split(';')[0]).join('; ')};
 }
 const login=(who,portal)=>request(portal==='student'?'/auth/login':portal==='client_admin'?'/auth/admin/login':'/auth/super-admin/login',{method:'POST',portal,body:{email:fixtures[who].email,password}});
 let studentCookie,otherCookie,adminCookie,builderCookie,requestId,invoiceId,secondPayment;
 await t.test('anonymous and malicious-origin requests are rejected',async()=>{assert.equal((await request('/business/overview')).status,401);assert.equal((await request('/auth/admin/login',{method:'POST',body:{},headers:{Origin:'https://attacker.invalid'}})).status,403);});
 await t.test('student login rejects both admin roles',async()=>{assert.equal((await login('client','student')).status,401);assert.equal((await login('builder','student')).status,401);});
 await t.test('valid students get independent HttpOnly sessions',async()=>{const a=await login('student','student'),b=await login('other','student');assert.equal(a.status,200);assert.equal(b.status,200);assert.match(a.headers.get('set-cookie'),/HttpOnly/i);assert.match(a.headers.get('set-cookie'),/SameSite=Lax/i);studentCookie=a.cookie;otherCookie=b.cookie;assert.equal((await request('/business/overview',{cookie:studentCookie})).status,403);});
 async function setupAdmin(who,portal){const first=await login(who,portal);assert.equal(first.status,200);assert.equal(first.data.setupRequired,true);assert.equal((await request('/admin/content/courses',{cookie:first.cookie,portal})).status,403);
  const setup=await request('/auth/security/mfa/setup',{method:'POST',cookie:first.cookie,portal,body:{currentPassword:password}});assert.equal(setup.status,200);
  const enable=await request('/auth/security/mfa/enable',{method:'POST',cookie:first.cookie,portal,body:{currentPassword:password,code:hotp(setup.data.secret,Math.floor(Date.now()/30000))}});assert.equal(enable.status,200);assert.equal(enable.data.recoveryCodes.length,8);
  assert.equal((await request('/admin/content/courses',{cookie:first.cookie,portal})).status,401);return enable.cookie;
 }
 await t.test('admin MFA setup is required and revokes the restricted session',async()=>{adminCookie=await setupAdmin('client','client_admin');builderCookie=await setupAdmin('builder','super_admin');});
 await t.test('builder can manage content, never the client ledger',async()=>{assert.equal((await request('/admin/content/courses',{cookie:builderCookie,portal:'super_admin'})).status,200);assert.equal((await request('/business/overview',{cookie:builderCookie,portal:'super_admin'})).status,403);assert.equal((await request('/business/overview',{cookie:builderCookie,portal:'client_admin'})).status,401);assert.equal((await request('/admin/site/footerSocial',{method:'PUT',body:{value:{}},cookie:adminCookie,portal:'client_admin'})).status,403);});
 await t.test('enrollment requests are durable without cash or access; duplicate is deduplicated',async()=>{const body={kind:'course',product:'integration-fixture',message:'Please arrange manual enrollment.'};const a=await request('/payments/requests',{method:'POST',cookie:studentCookie,body}),b=await request('/payments/requests',{method:'POST',cookie:studentCookie,body});assert.equal(a.status,201);assert.equal(b.status,200);requestId=a.data.item._id;assert.equal(await PurchaseRequest.countDocuments(),1);assert.equal(await Invoice.countDocuments(),0);assert.equal(await Payment.countDocuments(),0);assert.equal(await Enrollment.countDocuments(),0);});
 const day=new Date().toISOString().slice(0,10),due=new Date(Date.now()+10*86400000).toISOString().slice(0,10);
 const admin=(route,body,key=crypto.randomUUID())=>request(route,{method:'POST',portal:'client_admin',cookie:adminCookie,body:{...body,currentPassword:password,reason:'Integration validation only'},key});
 await t.test('request becomes one invoice but does not grant paid access',async()=>{const data={userId:String(fixtures.student._id),kind:'course',product:'integration-fixture',amount:'100.00',currency:'USD',startDate:day,dueDate:due,accessDays:365,purchaseRequestId:requestId};const key=crypto.randomUUID(),a=await admin('/invoices',data,key),b=await admin('/invoices',data,key);assert.equal(a.status,201,JSON.stringify(a.data));assert.equal(b.status,200,JSON.stringify(b.data));invoiceId=a.data.item._id;assert.equal(a.data.item._id,b.data.item._id);assert.equal(await Enrollment.countDocuments(),0);assert.equal((await PurchaseRequest.findById(requestId)).status,'invoiced');});
 await t.test('partial receipt changes balance but does not issue access',async()=>{const a=await admin('/payments',{invoiceId,amount:'30.00',method:'bank_transfer',reference:'integration-partial',receivedDate:day});assert.equal(a.status,201,JSON.stringify(a.data));assert.equal((await Invoice.findById(invoiceId)).paidMinor,3000);assert.equal(await Enrollment.countDocuments(),0);});
 await t.test('concurrent duplicate settlement cannot double charge the ledger or access',async()=>{const key=crypto.randomUUID(),body={invoiceId,amount:'70.00',method:'bank_transfer',reference:'integration-final',receivedDate:day};const results=await Promise.all([admin('/payments',body,key),admin('/payments',body,key)]);assert.ok(results.some(r=>r.status===201),JSON.stringify(results.map(r=>r.data)));assert.ok(results.every(r=>[200,201,409].includes(r.status)));assert.equal(await Payment.countDocuments({invoice:invoiceId}),2);assert.equal((await Invoice.findById(invoiceId)).paidMinor,10000);assert.equal(await Enrollment.countDocuments({user:fixtures.student._id}),1);secondPayment=await Payment.findOne({reference:'integration-final'});});
 await t.test('own billing cannot disclose another student invoices, even with injected user query',async()=>{const own=await request('/payments/mine/invoices',{cookie:studentCookie}),other=await request('/payments/mine/invoices?userId='+fixtures.student._id,{cookie:otherCookie});assert.equal(own.data.items.length,1);assert.equal(other.data.items.length,0);assert.ok(!JSON.stringify(own.data).includes('passwordHash'));});
 await t.test('completed offline refund creates credit without reopening debt',async()=>{const a=await admin('/refunds',{paymentId:String(secondPayment._id),amount:'10.00',reference:'integration-refund',refundedDate:day,revokeAccess:false});assert.equal(a.status,201,JSON.stringify(a.data));const i=await Invoice.findById(invoiceId);assert.equal(i.creditedMinor,1000);assert.equal(i.paidMinor,10000);assert.equal(await Refund.countDocuments(),1);});
 await t.test('forced real transaction exception rolls back a write',async()=>{await assert.rejects(()=>transaction(async session=>{await User.updateOne({_id:fixtures.other._id},{$set:{name:'SHOULD ROLL BACK'}},{session});throw new Error('Intentional rollback fixture');}),/Intentional rollback/);assert.equal((await User.findById(fixtures.other._id)).name,'other Test Fixture');});
 await t.test('customer blocking immediately invalidates active session',async()=>{const a=await admin('/customers/'+fixtures.student._id+'/actions',{action:'block'});assert.equal(a.status,200,JSON.stringify(a.data));assert.equal((await request('/payments/mine/invoices',{cookie:studentCookie})).status,401);});
});
